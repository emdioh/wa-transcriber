# wa-transcriber settings for the first boot.
#
# This file lives on the boot partition (FAT), so it can be edited from any
# computer: plug the SD card in, open the small FAT partition and edit the
# values below. Changes apply at the next boot.
#
# setup_ap_key: WPA2 passphrase of the setup access point "WA-Setup-XXXX"
#                (at least 8 characters; longer than 8 is fine, but some
#                phones reject passphrases over 63). If this line is missing
#                or too short, the device falls back to "wa-setup-<MAC>" and
#                logs a warning.
setup_ap_key=change-me-please

# country: ISO 3166-1 alpha-2 code for the WiFi regulatory domain.
country=IT

# wifi_ssid / wifi_passphrase: join this WiFi network on boot, without going
#                through the setup access point and the web UI. The network is
#                added to /data/wifi.json at start (the passphrase is turned
#                into the WPA key and is never stored) and shows up on the
#                Network page like one added there. The WPA passphrase must be
#                8 to 63 characters; a wrong one is logged as a warning and
#                ignored, and the device falls back to the setup AP. Leave
#                both lines commented out to configure the network in the UI.
# wifi_ssid=Casa
# wifi_passphrase=change-me-too

# hostname: mDNS name, <hostname>.local (default wa).
hostname=wa

# console: built-in read-only diagnostic console (no shell, no BusyBox).
#          "off" | "serial" | "usb" | "hdmi" | "both" (serial+usb) | "all"
#          (everything). "serial" is the UART header (/dev/ttyS0 on the
#          BeagleBone and Cubieboard, /dev/ttyAMA0 on the Raspberry Pi 1/2),
#          "usb" the USB client port CDC-ACM gadget (/dev/ttyGS0), "hdmi"
#          the HDMI output (/dev/tty1) with a USB keyboard.
#
#          Supports read-only inspection commands (help, uptime, mem, mounts,
#          ip, ps) plus reboot; there is no /bin/sh or arbitrary command
#          execution. Set "off" to disable completely. See os/README.md.
console=all
