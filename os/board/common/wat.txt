# wa-transcriber settings for the first boot.
#
# This file lives on the boot partition (FAT), so it can be edited from any
# computer: plug the SD card in, open the small FAT partition and edit the
# values below. Changes apply at the next boot.
#
# setup_ap_key: WPA2 passphrase of the setup access point "WA-Setup-XXXX"
#                (at least 8 characters; longer than 8 is fine, but some
#                phones reject passphrases over 63). If this line is missing
#                or too short, the device falls back to "wa-setup-<MAC>" and
#                logs a warning.
setup_ap_key=change-me-please

# country: ISO 3166-1 alpha-2 code for the WiFi regulatory domain.
country=IT

# hostname: mDNS name, <hostname>.local (default wa).
hostname=wa

# console: debug login shell on the device. "off" | "serial" | "usb" | "hdmi" |
#          "both" (serial+usb) | "all" (everything). "serial" is the UART
#          header (/dev/ttyS0 on the BeagleBone, /dev/ttyAMA0 on the Raspberry
#          Pi 2), "usb" the mini-USB client port (BeagleBone only), "hdmi" the
#          HDMI output with a USB keyboard on the host port.
#
#          These images ship with "all" while the boot and network work is
#          still being validated. There is no root password: anyone who can
#          reach one of those ports is root, so set "off" before the device
#          leaves your hands. See os/README.md.
console=all
